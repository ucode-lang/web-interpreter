# ucodepen -- build the ucode web interpreter as a WebAssembly module.
#
#   cmake -B build -DCMAKE_TOOLCHAIN_FILE=cmake/emscripten.cmake -DCMAKE_BUILD_TYPE=Release
#   cmake --build build
#
# ucode (and its json-c and libmd dependencies) is fetched from
# github.com/ucode-lang/ucode and built with its own CMake emscripten
# support; the web bridge is then linked against the resulting static
# libraries. Network access and an activated emsdk are required.

cmake_minimum_required(VERSION 3.15)
include(ExternalProject)

project(ucodepen-web C)

# Pinned ucode commit: the emscripten-support branch of
# github.com/ucode-lang/ucode. Bump this to track upstream.
set(UCODE_GIT_TAG b3182770c67d219d6b441decccd452d290f4e94c)

# Modules statically linked into ucode-modules.a and preloaded into the
# global `modules` dictionary at VM init. Must match what web-bridge.c
# expects in web-bridge-modules.inc.
set(WEB_MODULES io math struct fs zlib resolv socket digest)
# LIST_SEPARATOR below rewrites , -> ; in the external project's command
# lines, so the module list reaches ucode's CMake as one ;-separated value.
string(REPLACE ";" "," WEB_MODULES_CSV "${WEB_MODULES}")

# ---------------------------------------------------------------------------
# ucode: libucode.a + ucode-modules.a, plus json-c and libmd (which ucode's
# CMake emscripten support fetches and builds itself under <build>/deps/)
# ---------------------------------------------------------------------------
set(UCODE_PREFIX ${CMAKE_BINARY_DIR}/ucode)
set(UCODE_SRC_DIR ${UCODE_PREFIX}/src/ucode)
set(UCODE_BUILD_DIR ${UCODE_PREFIX}/src/ucode-build)

ExternalProject_Add(ucode-build
  PREFIX ${UCODE_PREFIX}
  SOURCE_DIR ${UCODE_SRC_DIR}
  BINARY_DIR ${UCODE_BUILD_DIR}
  GIT_REPOSITORY https://github.com/ucode-lang/ucode.git
  GIT_TAG ${UCODE_GIT_TAG}
  LIST_SEPARATOR ","
  CMAKE_ARGS
    -DCMAKE_TOOLCHAIN_FILE:FILEPATH=${CMAKE_CURRENT_SOURCE_DIR}/cmake/emscripten.cmake
    -DCMAKE_BUILD_TYPE:STRING=${CMAKE_BUILD_TYPE}
    -DUCODE_WASM:BOOL=ON
    -DUCODE_WASM_MODULES:STRING=${WEB_MODULES_CSV}
  UPDATE_COMMAND ""
  INSTALL_COMMAND ""
)

# The archives land in ucode's build directory (json-c and libmd in its
# deps/ external projects); the headers in the checked-out source tree.
add_library(ucode-lib STATIC IMPORTED GLOBAL)
set_property(TARGET ucode-lib PROPERTY IMPORTED_LOCATION ${UCODE_BUILD_DIR}/libucode.a)
add_library(ucode-modules-lib STATIC IMPORTED GLOBAL)
set_property(TARGET ucode-modules-lib PROPERTY IMPORTED_LOCATION ${UCODE_BUILD_DIR}/ucode-modules.a)
add_library(json-c-lib STATIC IMPORTED GLOBAL)
set_property(TARGET json-c-lib PROPERTY IMPORTED_LOCATION ${UCODE_BUILD_DIR}/deps/json-c/dist/lib/libjson-c.a)
add_library(libmd-lib STATIC IMPORTED GLOBAL)
set_property(TARGET libmd-lib PROPERTY IMPORTED_LOCATION ${UCODE_BUILD_DIR}/deps/libmd/dist/lib/libmd.a)

# ---------------------------------------------------------------------------
# Generated sources
# ---------------------------------------------------------------------------

# The module table for the web bridge, from WEB_MODULES.
set(MODULES_INC ${CMAKE_BINARY_DIR}/web-bridge-modules.inc)
set(_inc "/* Auto-generated by CMake -- do not edit. */\n\n")
string(APPEND _inc "#include <ucode/vm.h>\n")
string(APPEND _inc "#include <ucode/types.h>\n\n")
string(APPEND _inc "/* The renamed module init functions from ucode-modules.a. */\n")
foreach(mod ${WEB_MODULES})
  string(APPEND _inc "void uc_module_init_${mod}(uc_vm_t *vm, uc_value_t *scope);\n")
endforeach()
string(APPEND _inc "\ntypedef struct {\n")
string(APPEND _inc "\tconst char *name;\n")
string(APPEND _inc "\tvoid (*init)(uc_vm_t *, uc_value_t *);\n")
string(APPEND _inc "} web_module_entry_t;\n\n")
string(APPEND _inc "#define WEB_MODULES_TABLE")
set(_last "")
foreach(mod ${WEB_MODULES})
  set(_last ${mod})
endforeach()
foreach(mod ${WEB_MODULES})
  string(APPEND _inc " \\\n    { \"${mod}\", uc_module_init_${mod} }")
  if(NOT mod STREQUAL _last)
    string(APPEND _inc ",")
  endif()
endforeach()
string(APPEND _inc "\n")
file(WRITE ${MODULES_INC} "${_inc}")

# The demo filesystem (demo/ tree) as C string literals.
set(DEMO_INC ${CMAKE_BINARY_DIR}/web-bridge-demo.inc)
file(GLOB_RECURSE _demo_files ${CMAKE_CURRENT_SOURCE_DIR}/demo/*)
add_custom_command(
  OUTPUT ${DEMO_INC}
  COMMAND ${CMAKE_COMMAND}
          -DDEMO_SRC=${CMAKE_CURRENT_SOURCE_DIR}/demo
          -DDEMO_INC=${DEMO_INC}
          -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/gen-demo-inc.cmake
  DEPENDS ${_demo_files} ${CMAKE_CURRENT_SOURCE_DIR}/cmake/gen-demo-inc.cmake
  COMMENT "Generating web-bridge-demo.inc"
)

set_source_files_properties(${MODULES_INC} ${DEMO_INC}
  PROPERTIES GENERATED TRUE HEADER_FILE_ONLY TRUE)

# ---------------------------------------------------------------------------
# The web bridge (ucode.js + ucode.wasm)
# ---------------------------------------------------------------------------
add_executable(ucodepen
  src/web-bridge.c
  ${MODULES_INC}
  ${DEMO_INC}
)
add_dependencies(ucodepen ucode-build)

target_include_directories(ucodepen PRIVATE
  ${UCODE_SRC_DIR}/include
  ${UCODE_BUILD_DIR}/deps/json-c/dist/include
  ${CMAKE_BINARY_DIR}
)
target_compile_options(ucodepen PRIVATE
  -D_GNU_SOURCE
  -Wno-implicit-function-declaration
  -Wno-unused-parameter
)
target_link_libraries(ucodepen PRIVATE
  ucode-lib
  ucode-modules-lib
  json-c-lib
  libmd-lib
)

# emscripten link settings (emcc is the link driver under the toolchain).
set(_q "")
foreach(f _ucode_run _ucode_eval _ucode_run_file _ucode_render_file
          _ucode_render_string _ucode_reset _ucode_get_output
          _ucode_get_render _ucode_clear_output _ucode_clear_render
          _ucode_get_error _ucode_module_count _ucode_fs_seed_demo
          _ucode_fs_read _ucode_fs_run _ucode_fs_list _ucode_vfs_write
          _ucode_vfs_remove _ucode_vfs_clear _ucode_vfs_list
          _malloc _free)
  if(_q)
    string(APPEND _q ",")
  endif()
  string(APPEND _q "\"${f}\"")
endforeach()

target_link_options(ucodepen PRIVATE
  --pre-js=${CMAKE_CURRENT_SOURCE_DIR}/src/web-bridge.js
  -sUSE_ZLIB=1
  -sALLOW_MEMORY_GROWTH=1
  -sFORCE_FILESYSTEM=1
  "-sEXPORTED_FUNCTIONS=[${_q}]"
  "-sEXPORTED_RUNTIME_METHODS=[\"UTF8ToString\",\"lengthBytesUTF8\",\"stringToUTF8\"]"
  -sMODULARIZE=1
  -sEXPORT_NAME=ucodeWasm
  -sSTACK_SIZE=524288
  -sEXIT_RUNTIME=0
  -sNO_EXIT_RUNTIME=1
  -sASYNCIFY=0
)

# emcc with MODULARIZE emits <name>.js plus a sibling <name>.wasm.
set_target_properties(ucodepen PROPERTIES
  OUTPUT_NAME ucode
  SUFFIX .js
)

# Publish the runtime next to the web UI and into dist/ (the test suites
# load dist/ucode.js).
add_custom_command(TARGET ucodepen POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E make_directory
          ${CMAKE_CURRENT_SOURCE_DIR}/web ${CMAKE_CURRENT_SOURCE_DIR}/dist
  COMMAND ${CMAKE_COMMAND} -E copy
          $<TARGET_FILE:ucodepen> ${CMAKE_CURRENT_SOURCE_DIR}/web/ucode.js
  COMMAND ${CMAKE_COMMAND} -E copy
          ${CMAKE_CURRENT_BINARY_DIR}/ucode.wasm ${CMAKE_CURRENT_SOURCE_DIR}/web/ucode.wasm
  COMMAND ${CMAKE_COMMAND} -E copy
          $<TARGET_FILE:ucodepen> ${CMAKE_CURRENT_SOURCE_DIR}/dist/ucode.js
  COMMAND ${CMAKE_COMMAND} -E copy
          ${CMAKE_CURRENT_BINARY_DIR}/ucode.wasm ${CMAKE_CURRENT_SOURCE_DIR}/dist/ucode.wasm
  COMMENT "Publishing ucode.js + ucode.wasm to web/ and dist/"
)
